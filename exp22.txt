% Series
series(R1, R2, R) :- R is R1 + R2.

% Parallel
parallel(R1, R2, R) :- R is (R1 * R2) / (R1 + R2).

% Mixed circuit: R1 ani R2 parallel, tyala R3 series
equivalent(R1, R2, R3, R) :- parallel(R1, R2, Rp), series(Rp, R3, R).

query:
?- series(10, 20, R).
R = 30.

?- parallel(10, 10, R).
R = 5.

?- parallel(20, 30, R).
R = 12.

?- equivalent(10, 10, 5, R).
R = 10.
