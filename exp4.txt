% Maximum
max(X, Y, X) :- X >= Y.
max(X, Y, Y) :- X < Y.

% Minimum
min(X, Y, X) :- X =< Y.
min(X, Y, Y) :- X > Y.

querys:

?- max(10, 20, M).
M = 20.

?- min(10, 20, M).
M = 10.

?- max(50, 8, M).
M = 50.

?- min(50, 8, M).
M = 8.
